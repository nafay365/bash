sudo apt update
sudo apt install samba -y

sudo cp /etc/samba/smb.conf /etc/samba/smb.conf.backup

sudo nano /etc/samba/smb.conf


[Home]
   path = /home/sammers
   browseable = yes
   read only = no
   valid users = sammers
   create mask = 0755
   directory mask = 0755
   force user = sammers
   force group = sammers
   guest ok = no
   writable = yes

sudo smbpasswd -a sammers 

sudo systemctl restart smbd
